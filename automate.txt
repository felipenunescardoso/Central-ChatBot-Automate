-- ============================================================
-- PROJETO AUTOMATE - SOBREAVISO V1
-- Banco: automate
-- Schema: sobreaviso
-- PostgreSQL
--
-- Regras contempladas no modelo:
--   * Departamentos e usuários
--   * Perfis ADMIN / GESTOR / TECNICO
--   * Senha armazenada como hash SHA-256 com salt individual
--   * Cadastro de feriados
--   * Escalas com ordem e opção de replicação
--   * Divisão de uma escala por competência (dia 11 ao dia 10)
--   * Bloqueio/processamento de competências encerradas
--   * Lançamentos de SOBREAVISO e ACIONAMENTO
--   * Controle de geração/envio dos PDFs
--
-- IMPORTANTE:
-- SHA-256 é função de HASH, não criptografia reversível.
-- A senha em texto puro nunca é armazenada.
-- ============================================================

CREATE EXTENSION IF NOT EXISTS pgcrypto;

CREATE SCHEMA IF NOT EXISTS sobreaviso;

CREATE TABLE sobreaviso.departamentos
(
    id_departamento INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    nome_departamento VARCHAR(100) NOT NULL,
    email_responsavel_1 VARCHAR(200),
    email_responsavel_2 VARCHAR(200),
    ativo BOOLEAN NOT NULL DEFAULT TRUE,
    data_cadastro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uq_departamento_nome
        UNIQUE (nome_departamento)
);

CREATE TABLE sobreaviso.usuarios
(
    id_usuario INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_departamento INTEGER NOT NULL,
    nome VARCHAR(150) NOT NULL,
    chapa VARCHAR(30),
    email VARCHAR(200) NOT NULL,
    login VARCHAR(100) NOT NULL,

    senha_salt BYTEA NOT NULL,
    senha_hash CHAR(64) NOT NULL,

    perfil VARCHAR(20) NOT NULL,
    ativo BOOLEAN NOT NULL DEFAULT TRUE,
    data_cadastro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_usuario_departamento
        FOREIGN KEY (id_departamento)
        REFERENCES sobreaviso.departamentos(id_departamento),

    CONSTRAINT uq_usuario_login
        UNIQUE (login),

    CONSTRAINT ck_usuario_perfil
        CHECK (perfil IN ('ADMIN', 'GESTOR', 'TECNICO'))
);

CREATE TABLE sobreaviso.feriados
(
    id_feriado INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    data_feriado DATE NOT NULL,
    descricao VARCHAR(150) NOT NULL,
    ativo BOOLEAN NOT NULL DEFAULT TRUE,
    data_cadastro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uq_feriado_data
        UNIQUE (data_feriado)
);

CREATE TABLE sobreaviso.escalas
(
    id_escala INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_usuario INTEGER NOT NULL,
    data_hora_inicio TIMESTAMP NOT NULL,
    data_hora_fim TIMESTAMP NOT NULL,
    ordem INTEGER NOT NULL,
    replicacao BOOLEAN NOT NULL DEFAULT FALSE,
    id_escala_origem INTEGER,
    data_cadastro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_escala_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES sobreaviso.usuarios(id_usuario),

    CONSTRAINT fk_escala_origem
        FOREIGN KEY (id_escala_origem)
        REFERENCES sobreaviso.escalas(id_escala),

    CONSTRAINT ck_escala_datas
        CHECK (data_hora_fim > data_hora_inicio),

    CONSTRAINT ck_escala_ordem
        CHECK (ordem > 0)
);

CREATE TABLE sobreaviso.escala_competencias
(
    id_escala_competencia INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_escala INTEGER NOT NULL,
    competencia_inicio DATE NOT NULL,
    competencia_fim DATE NOT NULL,
    data_hora_inicio TIMESTAMP NOT NULL,
    data_hora_fim TIMESTAMP NOT NULL,
    bloqueada BOOLEAN NOT NULL DEFAULT FALSE,
    processada BOOLEAN NOT NULL DEFAULT FALSE,
    data_processamento TIMESTAMP,

    CONSTRAINT fk_escala_competencia_escala
        FOREIGN KEY (id_escala)
        REFERENCES sobreaviso.escalas(id_escala),

    CONSTRAINT ck_escala_competencia_datas
        CHECK (data_hora_fim > data_hora_inicio),

    CONSTRAINT ck_competencia_datas
        CHECK (competencia_fim >= competencia_inicio),

    CONSTRAINT uq_escala_competencia_periodo
        UNIQUE (id_escala, competencia_inicio, competencia_fim)
);

CREATE TABLE sobreaviso.lancamentos
(
    id_lancamento INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_escala_competencia INTEGER NOT NULL,
    tipo VARCHAR(20) NOT NULL,
    data_hora_inicio TIMESTAMP NOT NULL,
    data_hora_fim TIMESTAMP NOT NULL,
    motivo VARCHAR(1000),
    data_cadastro TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_lancamento_escala_competencia
        FOREIGN KEY (id_escala_competencia)
        REFERENCES sobreaviso.escala_competencias(id_escala_competencia),

    CONSTRAINT ck_lancamento_tipo
        CHECK (tipo IN ('SOBREAVISO', 'ACIONAMENTO')),

    CONSTRAINT ck_lancamento_datas
        CHECK (data_hora_fim > data_hora_inicio)
);

CREATE TABLE sobreaviso.envios
(
    id_envio INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_usuario INTEGER NOT NULL,
    competencia_inicio DATE NOT NULL,
    competencia_fim DATE NOT NULL,
    data_geracao TIMESTAMP,
    data_envio TIMESTAMP,
    email_usuario VARCHAR(200),
    email_copia_1 VARCHAR(200),
    email_copia_2 VARCHAR(200),
    nome_arquivo VARCHAR(300),
    status VARCHAR(20) NOT NULL DEFAULT 'PENDENTE',
    erro TEXT,

    CONSTRAINT fk_envio_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES sobreaviso.usuarios(id_usuario),

    CONSTRAINT ck_envio_status
        CHECK (status IN ('PENDENTE', 'GERADO', 'ENVIADO', 'ERRO')),

    CONSTRAINT ck_envio_competencia_datas
        CHECK (competencia_fim >= competencia_inicio),

    CONSTRAINT uq_envio_usuario_competencia
        UNIQUE (id_usuario, competencia_inicio, competencia_fim)
);

CREATE OR REPLACE FUNCTION sobreaviso.hash_senha
(
    p_senha TEXT,
    p_salt BYTEA
)
RETURNS CHAR(64)
LANGUAGE SQL
IMMUTABLE
STRICT
AS
$$
    SELECT encode(
        digest(
            p_salt || convert_to(p_senha, 'UTF8'),
            'sha256'
        ),
        'hex'
    )::CHAR(64);
$$;

CREATE OR REPLACE FUNCTION sobreaviso.criar_usuario
(
    p_id_departamento INTEGER,
    p_nome VARCHAR(150),
    p_chapa VARCHAR(30),
    p_email VARCHAR(200),
    p_login VARCHAR(100),
    p_senha TEXT,
    p_perfil VARCHAR(20)
)
RETURNS INTEGER
LANGUAGE plpgsql
AS
$$
DECLARE
    v_salt BYTEA;
    v_id_usuario INTEGER;
BEGIN
    IF p_senha IS NULL OR LENGTH(p_senha) = 0 THEN
        RAISE EXCEPTION 'A senha não pode ser vazia.';
    END IF;

    v_salt := gen_random_bytes(16);

    INSERT INTO sobreaviso.usuarios
    (
        id_departamento,
        nome,
        chapa,
        email,
        login,
        senha_salt,
        senha_hash,
        perfil
    )
    VALUES
    (
        p_id_departamento,
        p_nome,
        p_chapa,
        p_email,
        p_login,
        v_salt,
        sobreaviso.hash_senha(p_senha, v_salt),
        p_perfil
    )
    RETURNING id_usuario
    INTO v_id_usuario;

    RETURN v_id_usuario;
END;
$$;

CREATE OR REPLACE FUNCTION sobreaviso.validar_senha
(
    p_login VARCHAR(100),
    p_senha TEXT
)
RETURNS BOOLEAN
LANGUAGE SQL
STABLE
AS
$$
    SELECT EXISTS
    (
        SELECT 1
        FROM sobreaviso.usuarios u
        WHERE u.login = p_login
          AND u.ativo = TRUE
          AND u.senha_hash =
              sobreaviso.hash_senha(p_senha, u.senha_salt)
    );
$$;
